% Builds \TeamTabRows: the team as rows of a two-column table. Each name is
% wrapped in \teamcell{...}; with an odd number of names the last one is wrapped
% in \teamlast{...} (centred under both columns on the cover). Callers redefine
% these two macros (cover, declaration).
\newcommand{\TeamTabRows}{}
\newcommand{\TeamCount}{0}
\newcommand{\teamcell}[1]{#1}
\newcommand{\teamlast}[1]{\multicolumn{2}{c}{\teamcell{#1}}}
\newcommand{\buildteamrows}{%
  \gdef\TeamTabRows{}%
  \foreach \n [count=\i] in \TeamMembers {\xdef\TeamCount{\i}}%
  \foreach \n [count=\i] in \TeamMembers {%
    \ifodd\i
      \ifnum\i=\TeamCount
        \expandafter\gappto\expandafter\TeamTabRows\expandafter{\expandafter\teamlast\expandafter{\n}}%
      \else
        \expandafter\gappto\expandafter\TeamTabRows\expandafter{\expandafter\teamcell\expandafter{\n}&}%
      \fi
    \else
      \expandafter\gappto\expandafter\TeamTabRows\expandafter{\expandafter\teamcell\expandafter{\n}\\}%
    \fi}}
