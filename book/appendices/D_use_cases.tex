\chapter{Use Case Specifications}\label{app:use-cases}
This appendix specifies every use case of \ProjectName{}. The use-case diagrams are in \cref{chap:requirements};
each specification gives the actor, platform, pre- and postconditions, the main flow and the alternate or
exceptional flows. Identifiers are stable and are used by the traceability matrix (\cref{tab:rtm}).

\section{Account, Access and Identity}
\ucheading{UC-1}{Register with E-mail and Code}
\begin{tabularx}{\linewidth}{@{}>{\columncolor{najdapale}\bfseries\small}p{3.3cm}X@{}}
\toprule
Actor & Visitor (becomes Citizen); Brevo; Firebase Authentication \\ \midrule
Platform & Web, Mobile \\ \midrule
Preconditions & The visitor has no NAJDA account for the e-mail address. \\ \midrule
Postconditions (success) & A verified Firebase account and a NAJDA citizen profile exist, the latest Terms and Privacy Policy are accepted, and the user can sign in. \\ \midrule
Postconditions (failure) & Nothing exists in Firebase or in the database. \\ \bottomrule
\end{tabularx}
\ucflow{Main flow}
\begin{enumerate}[nosep,leftmargin=*,label=\textcolor{najdablue}{\bfseries\arabic*.}]
\item The visitor chooses Register and enters an e-mail address.
\item The system rate-limits the request, checks the address is free, and e-mails a six-digit code valid for 10 minutes (only a keyed hash is stored).
\item The visitor enters the code.
\item The system verifies it, and returns a one-time registration token valid for 15 minutes.
\item The visitor enters first and last name, phone number, address, gender and a password, and accepts the Terms and Privacy Policy.
\item The system validates the token, the e-mail and the phone number, creates the Firebase account (e-mail already verified), and in one database transaction saves the citizen, records the acceptance and consumes the verification; then it writes the role claim.
\item The client signs the user in and opens the home screen; the profile is already complete.
\end{enumerate}
\ucflow{Alternate flow --- e-mail already registered}
\begin{enumerate}[nosep,leftmargin=*,label=\textcolor{najdablue}{\bfseries\arabic*.}]
\item At step 2 the system answers with a conflict and the client offers to sign in instead.
\end{enumerate}
\ucflow{Alternate flow --- wrong, expired or exhausted code}
\begin{enumerate}[nosep,leftmargin=*,label=\textcolor{najdablue}{\bfseries\arabic*.}]
\item The system counts the failed attempt (kept even though the request fails). After five wrong attempts the visitor must request a new code; a new code can be requested only after the 60-second cool-down.
\end{enumerate}
\ucflow{Exception flow --- phone or password rejected}
\begin{enumerate}[nosep,leftmargin=*,label=\textcolor{najdablue}{\bfseries\arabic*.}]
\item A phone number already verified on another account, or a password that fails the Firebase password policy, is rejected before or without creating anything, with a message the visitor can act on.
\end{enumerate}
\ucflow{Exception flow --- database failure}
\begin{enumerate}[nosep,leftmargin=*,label=\textcolor{najdablue}{\bfseries\arabic*.}]
\item The newly created Firebase account is deleted again, so a retry starts clean.
\end{enumerate}

\begin{enumerate}[nosep,leftmargin=*,label=\textcolor{najdablue}{\bfseries\arabic*.}]

\begin{enumerate}[nosep,leftmargin=*,label=\textcolor{najdablue}{\bfseries\arabic*.}]

\end{enumerate}
\ucflow{Alternate flow --- in review}
\begin{enumerate}[nosep,leftmargin=*,label=\textcolor{najdablue}{\bfseries\arabic*.}]
\item A person at the provider is reviewing; the status moves to VERIFIED or DECLINED later.

\ucflow{Main flow}
\begin{enumerate}[nosep,leftmargin=*,label=\textcolor{najdablue}{\bfseries\arabic*.}]
\item The user enables App lock in Settings and authenticates once.
\item When the app returns from the background the lock screen appears.
\item The user authenticates with biometrics or the device credential to continue.
\end{enumerate}
\ucflow{Exception flow --- authentication fails}
\begin{enumerate}[nosep,leftmargin=*,label=\textcolor{najdablue}{\bfseries\arabic*.}]
\item The app stays locked.
\end{enumerate}
